# ChurchOS H5 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a complete, responsive ChurchOS requirement co-creation H5 and desktop website with working demo data flows for authentication, wishes, comments, sharing, notifications, campaign settings, and the admin dashboard.

**Architecture:** Keep the existing Node.js and Express backend with JSON-file persistence, refactor the server into a testable app factory, and replace the monolithic frontend script with focused ES modules. The browser uses a single application shell with explicit view and modal states; all writes go through REST APIs, while AI, email, geolocation, and translation remain deterministic demo adapters.

**Tech Stack:** Node.js 18+, Express 4, Multer, QRCode, Lucide, HTML5, CSS3, native ES modules, Node test runner, Supertest.

## Global Constraints

- Preserve the approved scope in `docs/superpowers/specs/2026-07-15-churchos-h5-design.md`.
- Keep the frontend framework-free; do not migrate to React, Vue, or a build tool.
- Use the existing JSON database under `db/` for local persistence.
- Keep real AI, email, SMS, OAuth, payment, and production database integrations outside this implementation.
- Validate at 375x812, 390x844, 430x932, 768x1024, and 1440x900.
- Use Lucide icons for interface actions and preserve only the ChurchOS cross as a custom brand mark.
- Keep card corner radii at 8px or less; modal shells may use 12px.
- Keep interactive controls at least 44px high on mobile.
- Never commit `node_modules/`, uploaded user files, generated exports, or browser screenshots.

```js
function createDatabase({ dir = path.join(__dirname, 'db') } = {}) {
  fs.mkdirSync(dir, { recursive: true });
  return { read, write, findById, insert, update, remove, seed };

  function filePath(table) {
    return path.join(dir, `${table}.json`);
  }

  function read(table) {
    const target = filePath(table);
    if (!fs.existsSync(target)) return [];
    return JSON.parse(fs.readFileSync(target, 'utf8') || '[]');
  }

  function write(table, rows) {
    fs.writeFileSync(filePath(table), JSON.stringify(rows, null, 2), 'utf8');
    return rows;
  }
}

const defaultDb = createDatabase();
module.exports = Object.assign(defaultDb, { createDatabase });
```

```js
// lib/http.js
function authenticate(database) {
  return (req, res, next) => {
    const match = /^Bearer\s+(\d+)$/.exec(req.headers.authorization || '');
    const user = match ? database.findById('users', Number(match[1])) : null;
    if (!user) return res.status(401).json({ error: '请先登录同工账号' });
    req.user = user;
    next();
  };
}

function adminAuthenticate(database) {
  const requireUser = authenticate(database);
  return (req, res, next) => requireUser(req, res, () => {
    if (!req.user.is_admin) return res.status(403).json({ error: '仅管理员可执行此操作' });
    next();
  });
}

module.exports = { authenticate, adminAuthenticate };
```

```js
const express = require('express');
const cors = require('cors');
const path = require('node:path');
const { createDatabase } = require('./db');

function createApp({ dbDir, uploadDir = path.join(__dirname, 'public', 'uploads') } = {}) {
  const app = express();
  const database = createDatabase({ dir: dbDir });
  database.seed();
  app.locals.db = database;
  app.locals.uploadDir = uploadDir;
  app.use(cors());
  app.use(express.json());
  app.use(express.urlencoded({ extended: true }));
  app.get('/api/health', (req, res) => res.json({ status: 'ok' }));
  app.use('/vendor/lucide', express.static(path.join(__dirname, 'node_modules/lucide/dist/umd')));
  app.use(express.static(path.join(__dirname, 'public')));
  app.use('/api', (req, res) => res.status(404).json({ error: '接口不存在' }));
  return app;
}

module.exports = { createApp };
```

```js
// public/js/api-client.mjs
export function formatApiError(payload, status) {
  return payload?.error || `请求失败 (${status})`;
}

export function createApiClient({ getToken, fetchImpl = fetch } = {}) {
  function authorizedHeaders(input) {
    const headers = new Headers(input || {});
    const token = getToken?.();
    if (token) headers.set('Authorization', `Bearer ${token}`);
    return headers;
  }

  async function request(path, options = {}) {
    const headers = authorizedHeaders(options.headers);
    if (options.body && !(options.body instanceof FormData)) headers.set('Content-Type', 'application/json');
    const response = await fetchImpl(path, { ...options, headers });
    const payload = response.headers.get('content-type')?.includes('application/json') ? await response.json() : await response.text();
    if (!response.ok) throw new Error(formatApiError(payload, response.status));
    return payload;
  }

  async function download(path, options = {}) {
    const response = await fetchImpl(path, { ...options, headers: authorizedHeaders(options.headers) });
    if (!response.ok) {
      const payload = await response.json().catch(() => ({}));
      throw new Error(formatApiError(payload, response.status));
    }
    return {
      blob: await response.blob(),
      filename: /filename=\"?([^\";]+)\"?/i.exec(response.headers.get('content-disposition') || '')?.[1] || 'download'
    };
  }

  return { request, download };
}
```

```js
// public/js/store.mjs
export function createStore(initialState) {
  let state = Object.freeze({ ...initialState });
  const listeners = new Set();
  return {
    initialState: state,
    get: () => state,
    set(patch) {
      state = Object.freeze({ ...state, ...patch });
      listeners.forEach((listener) => listener(state));
    },
    subscribe(listener) {
      listeners.add(listener);
      return () => listeners.delete(listener);
    }
  };
}
```

```js
}
```

```js
export function formatRemainingTime(deadline, now = Date.now()) {
  const remaining = Math.max(0, Date.parse(deadline) - now);
  const days = Math.floor(remaining / 86400000);
  const hours = Math.floor((remaining % 86400000) / 3600000);
  const minutes = Math.floor((remaining % 3600000) / 60000);
  const seconds = Math.floor((remaining % 60000) / 1000);
  return [days, hours, minutes, seconds].map((value) => String(value).padStart(2, '0')).join(':')
```
